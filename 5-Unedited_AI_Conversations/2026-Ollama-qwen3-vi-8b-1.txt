import json

INPUT_DIR = "~/ai_logs"  # Your raw chat logs (e.g., ~/ai_logs)
OUTPUT_DIR = "~/ai_archive"  # Where we store processed docs
GITHUB_REPO = "your_username/ai-tracker"  # Your GitHub repo

def process_log(log_path):
    """Turn raw chat log into structured document + index"""
    # Extract key: {input: "Your question", output: "AI response"}
    with open(log_path, "r") as f:
        log = f.read()
    
    # Format as JSON for indexing
    doc = {
        "id": f"{log_path.split('/')[-1]}_{datetime.now().strftime('%Y%m%d')}",
        "input": log.split("AI:")[0].strip(),  # Your prompt
        "output": log.split("AI:")[1].strip(), # Raw AI response
        "date": datetime.now().isoformat()
    }
    
    # Save to output dir with .json extension
    output_path = os.path.join(OUTPUT_DIR, f"{doc['id']}.json")
    with open(output_path, "w") as f:
        json.dump(doc, f, indent=2)
    
    return output_path

def auto_summarize():
    """Summarize ALL logs when total size > 50MB"""
    total_size = sum(os.path.getsize(f) for f in os.listdir(INPUT_DIR))
    if total_size > 50 * 1024 * 1024:  # 50MB
        print("⚠️ AUTO-SUMMARIZING: Total log size > 50MB!")
        subprocess.run([
            "git", "clone", GITHUB_REPO,
            f"--depth=1", f"--branch=auto_summarize"
        ])
        # Generate index from processed docs
        index = {"summary": "AI logs summarized by `ai-tracker`", "docs": []}
        for f in os.listdir(OUTPUT_DIR):
            if f.endswith(".json"):
                with open(os.path.join(OUTPUT_DIR, f), "r") as r:
                    index["docs"].append(json.load(r))
        # Push index to GitHub (1-click sync)
        subprocess.run([
            "git", "init", "-q", OUTPUT_DIR,
            "git", "add", "-A", OUTPUT_DIR,
            "git", "commit", "-m", "'Auto-summarized logs'",
            "git", "push", "origin", "auto_summarize"
        ])

def main():
    # 1. Process all raw chat logs (no AI needed here!)
    if not os.path.exists(OUTPUT_DIR):
        os.makedirs(OUTPUT_DIR)
    
    for log in os.listdir(INPUT_DIR):
        if log.endswith(".txt"):
            process_log(os.path.join(INPUT_DIR, log))
    
    # 2. Auto-summarize if needed
    auto_summarize()

if __name__ == "__main__":
    main()

import os
import json
import time
import re
from datetime import datetime

# 📌 CONFIGURATION - EDIT THESE PATHS
INPUT_DIR = r"C:\Users\thest\Desktop\Claude's Memory Ark"  # Your raw chat logs folder
OUTPUT_DIR = r"C:\Users\thest\Desktop\AI_Archive"  # Processed data
SUMMARY_DIR = r"C:\Users\thest\Desktop\AI_Summary"  # Automated summaries

# Ensure output directories exist
os.makedirs(OUTPUT_DIR, exist_ok=True)
os.makedirs(SUMMARY_DIR, exist_ok=True)

def process_log(log_path):
    """Convert raw chat logs to structured JSON format"""
    try:
        with open(log_path, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # Extract user prompt and AI response using pattern matching
        pattern = r'User:\s*([\s\S]+?)(?=AI:|$)'
        match = re.search(pattern, content, re.IGNORECASE)
        
        if not match:
            # Handle non-standard formats with fallback
            user_text = content.split("AI:")[0].strip()
            ai_text = content.split("AI:")[1].strip() if "AI:" in content else ""
        else:
            user_text = match.group(1).strip()
            ai_text = content.replace(f"User: {user_text}", "").strip()

        # Create structured document
        doc = {
            "id": f"{os.path.basename(log_path)}_{int(time.time())}",
            "user_prompt": user_text,
            "ai_response": ai_text,
            "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "file_path": log_path
        }
        
        # Save to structured output
        output_path = os.path.join(OUTPUT_DIR, f"{doc['id']}.json")
        with open(output_path, 'w', encoding='utf-8') as f:
            json.dump(doc, f, indent=2)
        
        return True
    except Exception as e:
        print(f"⚠️ Error processing {log_path}: {str(e)}")
        return False

def generate_summary():
    """Create a compressed summary document"""
    summary_lines = []
    total_size = 0
    
    # Build summary from processed data
    for filename in os.listdir(OUTPUT_DIR):
        if filename.endswith('.json'):
            try:
                with open(os.path.join(OUTPUT_DIR, filename), 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    summary_lines.append(f"📌 {data['timestamp']}\n")
                    summary_lines.append(f"Q: {data['user_prompt'][:100]}...\n")
                    summary_lines.append(f"A: {data['ai_response'][:100]}...\n")
                    summary_lines.append("-" * 40 + "\n")
                
                total_size += os.path.getsize(os.path.join(OUTPUT_DIR, filename))
            except:
                continue
    
    # Save summary with size tracking
    summary_path = os.path.join(SUMMARY_DIR, "ai_summary.txt")
    with open(summary_path, 'w', encoding='utf-8') as f:
        f.write("\n".join(summary_lines))
    
    print(f"✅ Generated summary at: {summary_path} (Total size: {total_size/1024:.1f}KB)")
    return total_size

def main():
    # Initial processing of all existing files
    processed_count = 0
    total_size = 0
    
    print("🚀 Processing existing chat logs...")
    for root, _, files in os.walk(INPUT_DIR):
        for file in files:
            if file.lower().endswith(('.txt', '.log', '.md')):
                if process_log(os.path.join(root, file)):
                    processed_count += 1
    
    # Generate initial summary
    if processed_count > 0:
        total_size = generate_summary()
        print(f"📊 Initial summary generated for {processed_count} logs")
    
    # Continuous monitoring mode
    print("🔍 Starting real-time monitoring...")
    while True:
        time.sleep(30)  # Check every 30 seconds
        
        # Check for new files
        new_files = []
        for root, _, files in os.walk(INPUT_DIR):
            for file in files:
                if file.lower().endswith(('.txt', '.log', '.md')):
                    new_files.append(os.path.join(root, file))
        
        # Process new files
        for new_file in new_files:
            if not new_file.startswith(OUTPUT_DIR):  # Avoid duplicates
                if process_log(new_file):
                    processed_count += 1
        
        # Check for summary trigger
        if processed_count > 0:
            total_size = generate_summary()
            print(f"💡 {processed_count} logs processed | Summary size: {total_size/1024:.1f}KB")

if __name__ == "__main__":
